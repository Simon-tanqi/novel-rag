#     python -m venv .ragas_venv
#     .\.ragas_venv\Scripts\python.exe -m pip install -r requirements-extras-ragas.txt

# 只用 RAGAS 的评测链路，不需要 torch CUDA：torch 走 CPU 源，体积小很多：
#     .\.ragas_venv\Scripts\python.exe -m pip install torch --index-url https://download.pytorch.org/whl/cpu
#     .\.ragas_venv\Scripts\python.exe -m pip install -r requirements-extras-ragas.txt
